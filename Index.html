Q1

ALGORITHM — CSC 123 COURSE REGISTRATION

Pseudocode Algorithm
  1. START
  2. DISPLAY "Welcome to PLASU Course Registration Portal"
  3. INPUT Matric Number
  4. INPUT Password
  5. IF Matric Number OR Password is empty THEN
  6.     DISPLAY "Error: Credentials cannot be empty" → go to step 3
  7. END IF
  8. DISPLAY "Login Successful — Loading Dashboard..."
  9. DISPLAY "Navigating to Course Registration Module"
  10. DISPLAY "Course Found: CSC 123 — Introduction to Computing"
  11. DISPLAY "Do you want to register CSC 123? (Y/N)"
  12. INPUT Choice
  13. IF Choice = "Y" THEN
  14.     DISPLAY "CSC 123 Successfully Registered!"
  15.     DISPLAY Registration Receipt with Matric Number
  16. ELSE
  17.     DISPLAY "Registration Cancelled"
  18. END IF
  19. STOP
🖥 BASIC — Q1 Source Code 
REM =============================================
REM  CSC ASSIGNMENT - QUESTION 1
REM  Course Registration for CSC 123
REM  Plateau State University Bokkos (PLASU)
REM =============================================

CLS
PRINT "======================================="
PRINT "  PLASU COURSE REGISTRATION PORTAL"
PRINT "======================================="
PRINT

REM --- Step 1: Get Login Credentials ---
INPUT "Enter Matric Number: ", MATRIC$
INPUT "Enter Password    : ", PASS$

REM --- Step 2: Validate Input ---
IF MATRIC$ = "" OR PASS$ = "" THEN
    PRINT "ERROR: Matric number and password are required!"
    END
END IF

REM --- Step 3: Login Success ---
PRINT
PRINT "Login Successful! Welcome, "; MATRIC$
PRINT "Navigating to Course Registration..."
PRINT

REM --- Step 4: Show Course ---
PRINT "Course Found:"
PRINT "  Code  : CSC 123"
PRINT "  Title : Introduction to Computing"
PRINT "  Units : 3"
PRINT
INPUT "Register this course? (Y/N): ", CHOICE$

REM --- Step 5: Register or Cancel ---
IF CHOICE$ = "Y" OR CHOICE$ = "y" THEN
    PRINT
    PRINT "======================================="
    PRINT "       REGISTRATION RECEIPT"
    PRINT "======================================="
    PRINT "Student : "; MATRIC$
    PRINT "Course  : CSC 123"
    PRINT "Status  : SUCCESSFULLY REGISTERED"
    PRINT "======================================="
ELSE
    PRINT "Registration Cancelled."
END IF

END
Q2

ALGORITHM — AVERAGE OF THREE NEGATIVE INTEGERS

Pseudocode Algorithm
  1. START
  2. DISPLAY "Enter three negative integers"
  3. INPUT A, B, C
  4. IF A ≥ 0 OR B ≥ 0 OR C ≥ 0 THEN
  5.     DISPLAY "Error: All integers must be negative" → go to step 2
  6. END IF
  7. Compute SUM = A + B + C
  8. Compute AVERAGE = SUM / 3
  9. DISPLAY "Sum =", SUM
  10. DISPLAY "Average =", AVERAGE
  11. STOP
🖥 BASIC — Q2 Source Code 
REM =============================================
REM  CSC ASSIGNMENT - QUESTION 2
REM  Average of Three Negative Integers
REM =============================================

CLS
PRINT "======================================="
PRINT "  AVERAGE OF THREE NEGATIVE INTEGERS"
PRINT "======================================="
PRINT

REM --- Step 1: Accept Input ---
INPUT "Enter first negative integer : ", A
INPUT "Enter second negative integer: ", B
INPUT "Enter third negative integer : ", C

REM --- Step 2: Validate (must all be negative) ---
IF A >= 0 OR B >= 0 OR C >= 0 THEN
    PRINT
    PRINT "ERROR: All three integers must be negative!"
    END
END IF

REM --- Step 3: Calculate ---
LET SUM = A + B + C
LET AVG = SUM / 3

REM --- Step 4: Display Results ---
PRINT
PRINT "======================================="
PRINT "           RESULTS"
PRINT "======================================="
PRINT "Integers : "; A; ","; B; ","; C
PRINT "Sum      : "; SUM
PRINT "Average  : "; AVG
PRINT "======================================="

END
